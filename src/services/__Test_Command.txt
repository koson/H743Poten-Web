// Step 3: Clear previous measurement and start fresh
console.log("🔍 Step 3: Clearing Previous Data");

async function clearAndStartFresh() {
    try {
        console.log("🧹 3.1: Clearing previous measurement data...");
        
        // Try to clear/reset STM32 state
        const clearCommands = [
            "POTEn:CV:ABORt",  // Abort any running measurement
            "SYSTem:CLEar",    // Clear system state (if supported)
            "*RST"             // Reset device (if supported)
        ];
        
        for (const cmd of clearCommands) {
            try {
                console.log(`   Trying: ${cmd}`);
                const response = await fetch('/api/uart/send', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ command: cmd })
                });
                const result = await response.json();
                console.log(`   Response: ${result.response || result.error}`);
                
                // Wait between commands
                await new Promise(resolve => setTimeout(resolve, 1000));
            } catch (e) {
                console.log(`   ${cmd} failed:`, e.message);
            }
        }
        
        // Wait a bit for STM32 to process
        console.log("⏳ Waiting 3 seconds for STM32 to reset...");
        await new Promise(resolve => setTimeout(resolve, 3000));
        
        // Check status after clearing
        console.log("📊 3.2: Checking status after clear...");
        const statusResponse = await fetch('/api/uart/send', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ command: 'POTEn:CV:STATUS?' })
        });
        const statusResult = await statusResponse.json();
        console.log("   Status after clear:", statusResult.response);
        
        await new Promise(resolve => setTimeout(resolve, 1000));
        
        // Check data after clearing
        console.log("📊 3.3: Checking data after clear...");
        const dataResponse = await fetch('/api/uart/send', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ command: 'POTEn:CV:DATA?' })
        });
        const dataResult = await dataResponse.json();
        console.log("   Data after clear:", dataResult.response?.length || 0, "chars");
        console.log("   Data content:", dataResult.response || "No data");
        
        // Analysis
        if (statusResult.response?.trim() === "IDLE") {
            console.log("✅ Step 3 PASSED - STM32 cleared and ready");
            console.log("📋 Ready for Step 4: Fresh CV Start");
            return true;
        } else {
            console.log("⚠️  Step 3 PARTIAL - Status still:", statusResult.response);

            console.log("📋 Ready for Step 4: Fresh CV Start");
            return true;

        return false;

// Step 4: Start Fresh CV Measurement
console.log("🔍 Step 4: Fresh CV Start");

async function startFreshCV() {
    try {
        console.log("🚀 4.1: Starting fresh CV measurement...");
        
        // Use different parameters to ensure fresh start
        const cvCommand = "POTEn:CV:Start:ALL -0.3,0.3,-0.3,0.1,1";
        console.log(`   Command: ${cvCommand}`);
        console.log("   Parameters: -0.3V to +0.3V, start at -0.3V, 0.1V/s, 1 cycle");
        
        const startResponse = await fetch('/api/uart/send', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ command: cvCommand })
        });
        const startResult = await startResponse.json();
        console.log("   Start Response:", startResult.response || startResult.error);
        
        if (!startResult.response || !startResult.response.includes("OK")) {
            console.log("❌ Step 4.1 FAILED - STM32 did not start CV");
            return false;
        }
        
        console.log("✅ Step 4.1 PASSED - CV started successfully");
        
        // Wait for STM32 to begin measurement
        console.log("⏳ 4.2: Waiting 2 seconds for measurement to begin...");
        await new Promise(resolve => setTimeout(resolve, 2000));
        
        // Check status during measurement
        console.log("📊 4.3: Checking status during measurement...");
        const statusResponse = await fetch('/api/uart/send', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ command: 'POTEn:CV:STATUS?' })
        });
        const statusResult = await statusResponse.json();
        console.log("   Status during measurement:", statusResult.response);
        
        await new Promise(resolve => setTimeout(resolve, 1000));
        
        // Check early data
        console.log("📊 4.4: Checking early data...");
        const dataResponse = await fetch('/api/uart/send', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ command: 'POTEn:CV:DATA?' })
        });
        const dataResult = await dataResponse.json();
        const dataLength = dataResult.response?.length || 0;
        console.log("   Early data length:", dataLength, "chars");
        
        if (dataLength > 50) {  // Substantial data
            const lines = dataResult.response.split('\n').filter(line => line.trim());
            console.log("   Data lines:", lines.length);
            console.log("   First line:", lines[0] || "No lines");
            console.log("   Second line:", lines[1] || "No second line");
            console.log("   Last line:", lines[lines.length - 1] || "No last line");
        } else {
            console.log("   Data preview:", dataResult.response || "No data");
        }
        
        // Analysis
        if (statusResult.response?.includes("MEASURING")) {
            console.log("🎉 Step 4 EXCELLENT - STM32 is actively measuring!");
            console.log("📋 Ready for Step 5: Real-time Monitoring");
            return "MEASURING";
        } else if (statusResult.response?.includes("COMPLETE")) {
            console.log("⚡ Step 4 FAST - Measurement completed quickly");
            if (dataLength > 100) {
                console.log("✅ And we have substantial data!");
                console.log("📋 Ready for Step 6: Data Analysis");
                return "COMPLETE_WITH_DATA";
            } else {
                console.log("⚠️  But data is still limited");
                console.log("📋 This confirms the 5-point issue");
                return "COMPLETE_LIMITED";
            }
        } else {
            console.log("❓ Step 4 UNCLEAR - Unexpected status:", statusResult.response);
            console.log("📋 Will proceed to monitoring anyway");
            return "UNCLEAR";
        }
        
    } catch (error) {
        console.log("❌ Step 4 FAILED - Start error:", error);
        return false;
    }
}

startFreshCV();

                const data = result.response || result.error || "";
                
                console.log(`   Response length: ${data.length} chars`);
                console.log(`   Raw response: "${data}"`);
                
                // Analyze response type
                if (data.includes("CDC:") || data.includes("Received") || data.includes("bytes:")) {

        console.log("This explains why web interface gets mixed debug + SCPI data.");
        
        console.log("\n💡 SOLUTIONS:");
        console.log("1. FIRMWARE: Disable debug output on SCPI channel");
        console.log("2. FIRMWARE: Separate debug channel from SCPI channel");
